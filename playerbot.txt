# Обзор
# "Randombot": случайно генерируемые боты, которые входят в систему отдельно от игроков и населяют мир. Случайные роботы могут автоматически гриндить, выполнять квесты, повышать уровень и улучшать оборудование, а также их можно приглашать в группы и давать им команды.
# "AddClass bot": боты из учетных записей AddClassAccountPoolSize. Они используются для быстрого добавления в вашу группу уровневого и экипированного бота любого класса. Их рекомендуют для быстрого формирования партии.
# "Altbot": персонажи, созданные в учетных записях игроков, в которые игрок может входить, приглашать их в группы и давать им команды, например, случайные боты. Они лучше всего подходят для длительных прохождений.
# Информацию о командах для управления ботами и настройке их стратегий можно найти на вики по адресу https://github.com/mod-playerbots/mod-playerbots/wiki/Playerbot-Commands..

# Включить или отключить модуль Playerbots
AiPlayerbot.Enabled = 1

# Включить систему случайных ботов
AiPlayerbot.RandomBotAutologin = 1

# Случайное количество ботов
AiPlayerbot.MinRandomBots = 500
AiPlayerbot.MaxRandomBots = 500

# Случайные аккаунты ботов
# Установите значение 0, чтобы автоматически рассчитывать необходимые учетные записи на основе MaxRandomBots, EnablePeriodidicOnlineOffline.
# и его соотношение, а также AddClassAccountPoolSize. Установите вручную, если вы хотите создать больше учетных записей, чем необходимо.
# Если заданное вручную значение меньше требуемого, система заменит его требуемым значением.
# По умолчанию: 0 (автоматически)
AiPlayerbot.RandomBotAccountCount = 0

# Добавить классные аккаунты
# Количество учетных записей, созданных для ботов, зарезервированных для команды addclass. Как и в случае со случайными ботами, каждая учетная запись имеет
# 10 ботов (или 9, если WotLK отключен), по одному боту на каждый класс.
AiPlayerbot.AddClassAccountPoolSize = 50

# Префикс для созданных аккаунтов ботов (любого типа).
# Не меняйте этот префикс, пока существуют учетные записи ботов.
AiPlayerbot.RandomBotAccountPrefix = "rndbot"

# Удалить все аккаунты рандомных ботов
# Чтобы применить это, установите число 1 и запустите Worldserver. После завершения удаления, если вы хотите
# хотите воссоздать случайных ботов, установите число обратно на 0 и перезапустите Worldserver.
AiPlayerbot.DeleteRandomBotAccounts = 0

# Отключить случайных ботов, когда ни один реальный игрок не авторизован.
# По умолчанию: 0 (случайные боты будут входить в систему при запуске сервера)
# Если эта опция включена, рандомные боты будут входить в систему только через 30 секунд (по умолчанию) после входа в систему реального игрока.
# выйти из системы через 300 секунд (по умолчанию) после выхода всех реальных игроков
AiPlayerbot.DisabledWithoutRealPlayer = 0
AiPlayerbot.DisabledWithoutRealPlayerLoginDelay = 30
AiPlayerbot.DisabledWithoutRealPlayerLogoutDelay = 300

#
# Две независимые дополнительные функции для управления случайными уровнями ботов с течением времени. Оба забрать
# изменены значения стандартной команды ".reload config" GM.
#
#   - Брекеты уровней: периодически перераспределяет случайных ботов по настраиваемому уровню каждой фракции.
#     скобки, при необходимости взвешивая распределение в сторону скобок, содержащих реальные (не ботовые)
#     игроки. Это дополняет поведение AiPlayerbot.DowngradeMaxLevelBot-style: вместо
#     однократное понижение версии, постоянно перебалансирует всю случайную популяцию ботов в скобках
#     через некоторое время.
#
#   - Сброс уровня: когда случайный бот достигает настроенного максимального уровня, он сбрасывает его обратно на более низкий уровень.
#     уровень через полную рандомизацию PlayerbotFactory с дополнительным процентным шансом (необязательно
#     масштабируется по уровню), необязательное ограничение минимального игрового времени и необязательный "skip", который
#     отправляет бота прямо на другой уровень, как только он достигает настроенного уровня триггера.
#
# ПРИМЕЧАНИЕ. Сброс уровня совпадает с AiPlayerbot.DowngradeMaxLevelBot, который понижает версию бота,
#       уже на RandomBotMaxLevel на RandomBotMinLevel в следующий раз периодически
#       повторно рандомизированы. Вместо этого сброс уровня срабатывает немедленно (или по таймеру игрового времени), как только
#       бот достигает настроенного уровня со своими собственными элементами управления chance/played-time. Обычно не
#       включите AiPlayerbot.DowngradeMaxLevelBot и AiPlayerbot.ResetBotLevel.Enabled в
#       в то же время.
#

# игроки, присутствующие в каждой группе уровней. Это переопределяет любые пользовательские скобки в файле conf.
# По умолчанию: 0 (отключено)
AiPlayerbot.LevelBrackets.Dynamic.UseDynamicDistribution = 0

# Определяет, какой дополнительный вес придается скобкам (диапазонам уровней), содержащим реальные (не боты)
# игроков, когда включено динамическое распределение. Чем выше это значение, тем больше ботов переходят в
# скобки того же уровня, где находятся настоящие игроки, но эффект мягкий, а не чрезмерный. Значение
# Значение 0,0 означает, что боты всегда распределяются равномерно по всем группам, независимо от того, где находятся игроки.
# Значение по умолчанию 1,0 дает мягкий, сбалансированный эффект. Повышение этого значения до 3,0, 5,0 или выше приводит к
# боты больше концентрируются в скобках с реальными игроками. Значение представляет собой множитель (не процент):
# 0,0 = никакого дополнительного эффекта, 1,0 = минимальный, 3,0 = более сильный, 5,0 = сильный, но не чрезмерный. Эксперимент
# на основе общего количества ботов и реального количества игроков, чтобы найти подходящее количество для вашего сервера. Если
# если вам нужно большое скопление ботов на вашем уровне для одиночной игры, рекомендуется 10-15.
# Чего ожидать:
#   - При версии 1.0 (по умолчанию): если 6/10 реальных игроков находятся в одной группе, эта группа получает около 12,77%
#     ботов. Полностью пустые скобки получают примерно по 10,69% каждая.
#   - С 3.0: тот же сценарий: группа получает около 15,73% ботов. Полностью пустые скобки: по 9,93% каждая.
#   - В версии 5.0: группа с 6/10 реальными игроками получает 18,31%. Полностью пустые скобки: по 9,28% каждая.
#   - Если все реальные игроки находятся в одной группе и вес равен 5,0, эта группа получает около 21,56%
#     боты (остальные: по 9,80%).
#   - При 0.0: в каждой группе всегда одинаковое количество ботов (e.g., по 11,11% за 9
#     скобки).
# Формула (на скобку):
#   bracket_weight = 1.0 + (RealPlayerWeight x (1 / TotalRealPlayers) x log(1 + RealPlayersInBracket))
#   Все веса кронштейнов нормализованы до 100%.
# По умолчанию: 1,0

# Если эта опция включена, и Альянс, и Орда должны иметь одинаковые определения скобок (одинаковое число, одинаковые
# границы уровня). Все реальные игроки (независимо от фракции) влияют на динамическое распределение
# обе фракции.
# По умолчанию: 0 (отключено)
# ПРИМЕЧАНИЕ. Если скобки не совпадают при включении, SyncFactions для этого принудительно отключается.
#       сеанс и записывается ошибка; сервер обязательно запустится заново.

# БотActiveAlone
# - Контролирует количество активных ботов, когда поблизости нет реальных игроков.
# - Подумайте об этом как о приблизительном проценте: 10 означает, что примерно 10% ботов будут активны.
#   Не точно — фактическое количество может незначительно меняться в зависимости от цикла вращения.
# - Активные боты меняются: каждые <DurationSeconds> чередуется другой набор ботов.
# - Реальное количество активных ботов всегда будет выше этого значения, поскольку боты в
#   бой, подземелья, поля боя, очередь LFG, группы с реальными игроками и т. д. всегда доступны.
#   помимо этого принудительно активен (см. правила принудительной активации ниже).
# - Установите значение 100 (с выключенным SmartScale) = все боты всегда активны. Максимальная нагрузка на сервер.
# - Установите значение 0 = активными будут только боты, соответствующие приведенному ниже правилу.
#
# BotActiveAloneDurationSeconds
# - Как часто меняется активный состав (в секундах). Просыпается другая группа ботов
#   и предыдущая группа может простаивать.
# - Это минимум, а не точно. Если бот находится в бою или соответствует какому-либо силовому правилу, когда
#   происходит вращение, оно остается активным до тех пор, пока эти условия не закончатся — оно не будет отключено
#   в середине боя только потому, что его очередь истекла.
#
AiPlayerbot.BotActiveAlone = 10
AiPlayerbot.BotActiveAloneDurationSeconds = 30

# Принудительно активные правила (1 = вкл., 0 = выкл.)
# Они переопределяют процент, указанный выше. Если любое из этих условий верно, бот остается активным.
#
# ИнРадиус  - Настоящий игрок находится в пределах указанного количества ярдов (для отключения установите значение 0).
# ИнЗоне    - Реальный игрок находится в той же зоне (e.g. Элвиннский лес).
# InMap     - Реальный игрок находится на том же континенте (e.g. Восточные королевства).
# Друг  - У настоящего игрока этот бот есть в списке друзей.
# В Гильдии   - Этот бот находится в гильдии, в которой есть реальный игрок.
#
# Боты также всегда активны (не настраиваемы), когда:
#   в бою, внутри dungeon/raid/BG, в очереди BG или LFG,
#   сгруппированы с реальным игроком или контролируются реальным игроком.
#
AiPlayerbot.BotActiveAloneForceWhenInRadius = 150
AiPlayerbot.BotActiveAloneForceWhenInZone = 1
AiPlayerbot.BotActiveAloneForceWhenInMap = 0
AiPlayerbot.BotActiveAloneForceWhenIsFriend = 0
AiPlayerbot.BotActiveAloneForceWhenInGuild = 1

# SmartScale — автоматически уменьшает количество активных ботов, когда на сервере возникают проблемы.
# Отслеживает время обновления сервера (сколько времени занимает каждый такт сервера в миллисекундах).
# Когда сервер замедляется, меньше ботов остается активными, чтобы снизить нагрузку.
#
#   Этаж (по умолчанию 50 мс)   - Ниже этого снижения нет. Сервер работает нормально.
#   Потолок (по умолчанию 200 мс). При достижении этого значения или выше все непринудительные боты приостанавливаются.
#   Между полом и потолком активность постепенно снижается.
#     Пример: BotActiveAlone=10, пол=50, потолок=200.
#       Сервер на 50 мс  → ~10% активно (без снижения)
#       Сервер на 125 мс → активен ~5% (снижение вдвое)
#       Сервер на 200мс → 0% активен (остаются только форсированные боты)
#
# MinLevel/MaxLevel — SmartScale затрагивает только ботов в этом диапазоне уровней.
#   Боты за пределами диапазона всегда используют полное значение BotActiveAlone.
#   Принудительные правила всегда побеждают SmartScale.
#
AiPlayerbot.botActiveAloneSmartScale = 1
AiPlayerbot.botActiveAloneSmartScaleDiffLimitfloor = 50
AiPlayerbot.botActiveAloneSmartScaleDiffLimitCeiling = 200
AiPlayerbot.botActiveAloneSmartScaleWhenMinLevel = 1
AiPlayerbot.botActiveAloneSmartScaleWhenMaxLevel = 80
